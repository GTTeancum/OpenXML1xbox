cmake_minimum_required(VERSION 3.20)
project(XML1DX8 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Build the genuine D3D8 renderer with -A Win32")
endif()
add_executable(xml1-dx8-worker dx8_worker.cpp)
target_compile_features(xml1-dx8-worker PRIVATE cxx_std_17)
target_include_directories(xml1-dx8-worker PRIVATE ../external/dx8-headers)
target_compile_definitions(xml1-dx8-worker PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_link_libraries(xml1-dx8-worker PRIVATE user32)
add_library(xml1-pc-controls STATIC ../src/pc_controls.cpp ../src/pc_input_channel.cpp ../src/pc_options_model.cpp)
target_sources(xml1-pc-controls PRIVATE ../src/pc_gamepad.cpp)
target_sources(xml1-pc-controls PRIVATE ../src/pc_native_options.cpp)
target_compile_features(xml1-pc-controls PUBLIC cxx_std_17)
target_compile_definitions(xml1-pc-controls PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(xml1-pc-controls PUBLIC ../src)
target_link_libraries(xml1-dx8-worker PRIVATE xml1-pc-controls)
add_executable(pc-controls-test ../tests/pc_controls_test.cpp)
target_link_libraries(pc-controls-test PRIVATE xml1-pc-controls)
add_executable(dx8-readback-test ../tests/dx8_readback_test.cpp)
target_compile_features(dx8-readback-test PRIVATE cxx_std_17)
target_include_directories(dx8-readback-test PRIVATE ../external/dx8-headers)
target_compile_definitions(dx8-readback-test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(dx8-readback-test PRIVATE user32)
